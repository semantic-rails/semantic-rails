SELECT
  metric_time__month
  , SUM(new_customer_orders) AS new_customer_orders
FROM (
  SELECT
    DATE_TRUNC('month', ordered_at) AS metric_time__month
    , CASE WHEN is_new_customer_order THEN 1 ELSE 0 END AS new_customer_orders
  FROM "jaffle_comparison"."main"."orders" orders_src_10000
) subq_2
GROUP BY
  metric_time__month
ORDER BY metric_time__month
