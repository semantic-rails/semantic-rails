SELECT
  subq_1.metric_time__month AS metric_time__month
  , stores_src_10000.store_name AS store__store_name
  , SUM(subq_1.revenue_usd) AS revenue_usd
FROM (
  SELECT
    DATE_TRUNC('month', ordered_at) AS metric_time__month
    , store_id AS store
    , order_total_cents / 100.0 AS revenue_usd
  FROM "jaffle_comparison"."main"."orders" orders_src_10000
) subq_1
LEFT OUTER JOIN
  "jaffle_comparison"."main"."stores" stores_src_10000
ON
  subq_1.store = stores_src_10000.store_id
GROUP BY
  subq_1.metric_time__month
  , stores_src_10000.store_name
ORDER BY metric_time__month
