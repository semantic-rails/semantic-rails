  SELECT
    store_name
    , store_id AS store
  FROM "jaffle_comparison"."main"."stores" stores_src_10000
)

  , MAX(subq_8.orders) AS orders
  , MAX(subq_17.item_revenue_usd) AS item_revenue_usd
FROM (
  SELECT
    subq_1.metric_time__month AS metric_time__month
    , rss_10007_cte.store_name AS store__store_name
    , SUM(subq_1.__orders) AS orders
  FROM (
    SELECT
      DATE_TRUNC('month', ordered_at) AS metric_time__month
      , store_id AS store
      , 1 AS __orders
    FROM "jaffle_comparison"."main"."orders" orders_src_10000
  ) subq_1
  LEFT OUTER JOIN
    rss_10007_cte
  ON
    subq_1.store = rss_10007_cte.store
  GROUP BY
    subq_1.metric_time__month
    , rss_10007_cte.store_name
) subq_8
FULL OUTER JOIN (
  SELECT
    subq_10.metric_time__month AS metric_time__month

    , SUM(subq_10.__item_revenue_usd) AS item_revenue_usd
  FROM (
    SELECT
      DATE_TRUNC('month', ordered_at) AS metric_time__month
      , store_id AS store
      , item_revenue_cents / 100.0 AS __item_revenue_usd
    FROM "jaffle_comparison"."main"."order_items" order_items_src_10000
  ) subq_10
  LEFT OUTER JOIN
    rss_10007_cte

    subq_10.store = rss_10007_cte.store

    subq_10.metric_time__month
