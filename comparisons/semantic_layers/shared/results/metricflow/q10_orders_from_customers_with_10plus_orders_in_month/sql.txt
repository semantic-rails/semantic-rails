SELECT
  metric_time__month
  , SUM(qualifying_orders) AS qualifying_orders
FROM (
  SELECT
    DATE_TRUNC('month', ordered_at) AS metric_time__month
    , 1 AS qualifying_orders
  FROM "jaffle_comparison"."main"."high_frequency_orders" high_frequency_orders_src_10000
) subq_2
GROUP BY
  metric_time__month
ORDER BY metric_time__month
