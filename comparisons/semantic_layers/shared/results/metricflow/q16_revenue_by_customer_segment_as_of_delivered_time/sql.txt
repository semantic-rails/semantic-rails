SELECT
  subq_1.metric_time__month AS metric_time__month
  , customer_history_src_10000.customer_segment AS customer__customer_segment
  , SUM(subq_1.delivered_revenue) AS delivered_revenue
FROM (
  SELECT
    DATE_TRUNC('day', delivered_at) AS metric_time__day
    , DATE_TRUNC('month', delivered_at) AS metric_time__month
    , customer_id AS customer
    , order_total_cents / 100.0 AS delivered_revenue
  FROM "jaffle_comparison"."main"."order_lifecycle" order_lifecycle_src_10000
) subq_1
LEFT OUTER JOIN
  "jaffle_comparison"."main"."customer_history" customer_history_src_10000
ON
  (
    subq_1.customer = customer_history_src_10000.customer_id
  ) AND (
    (
      subq_1.metric_time__day >= customer_history_src_10000.valid_from
    ) AND (
      (
        subq_1.metric_time__day < customer_history_src_10000.valid_to
      ) OR (
        customer_history_src_10000.valid_to IS NULL
      )
    )
  )
GROUP BY
  subq_1.metric_time__month
  , customer_history_src_10000.customer_segment
ORDER BY metric_time__month
