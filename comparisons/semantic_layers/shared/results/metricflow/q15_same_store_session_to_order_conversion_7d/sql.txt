SELECT
  metric_time__month
  , CAST(same_store_converted_sessions_7d AS DOUBLE) / CAST(NULLIF(session_starts, 0) AS DOUBLE) AS same_store_session_to_order_conversion_rate_7d
FROM (
  SELECT
    COALESCE(subq_4.metric_time__month, subq_9.metric_time__month) AS metric_time__month
    , MAX(subq_4.same_store_converted_sessions_7d) AS same_store_converted_sessions_7d
    , MAX(subq_9.session_starts) AS session_starts
  FROM (
    SELECT
      DATE_TRUNC('month', started_at) AS metric_time__month
      , SUM(same_store_converted_sessions_7d) AS same_store_converted_sessions_7d
    FROM "jaffle_comparison"."main"."session_conversions_7d_same_store" session_conversions_7d_same_store_src_10000
    GROUP BY
      DATE_TRUNC('month', started_at)
  ) subq_4
  FULL OUTER JOIN (
    SELECT
      metric_time__month
      , SUM(session_starts) AS session_starts
    FROM (
      SELECT
        DATE_TRUNC('month', started_at) AS metric_time__month
        , 1 AS session_starts
      FROM "jaffle_comparison"."main"."storefront_sessions" storefront_sessions_src_10000
    ) subq_7
    GROUP BY
      metric_time__month
  ) subq_9
  ON
    subq_4.metric_time__month = subq_9.metric_time__month
  GROUP BY
    COALESCE(subq_4.metric_time__month, subq_9.metric_time__month)
