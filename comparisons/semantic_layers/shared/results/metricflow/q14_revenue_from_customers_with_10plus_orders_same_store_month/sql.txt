WITH sma_10003_cte AS (
  SELECT
    DATE_TRUNC('month', ordered_at) AS metric_time__month
    , store_id AS store
    , customer_id || '|' || store_id || '|' || strftime(ordered_at, '%Y-%m') AS customer_store_month
    , 1 AS __orders
    , order_total_cents / 100.0 AS __qualifying_revenue_usd_same_store
  FROM "jaffle_comparison"."main"."orders" orders_src_10000
)

SELECT
  metric_time__month AS metric_time__month
  , store__store_name AS store__store_name
  , SUM(qualifying_revenue_usd_same_store) AS qualifying_revenue_usd_same_store
FROM (
  SELECT
    subq_5.metric_time__month AS metric_time__month
    , subq_13.store_name AS store__store_name
    , subq_11.customer_store_month__orders AS customer_store_month__orders
    , subq_5.__qualifying_revenue_usd_same_store AS qualifying_revenue_usd_same_store
  FROM (
    SELECT
      metric_time__month
      , store
      , customer_store_month
      , __qualifying_revenue_usd_same_store
    FROM sma_10003_cte
  ) subq_5
  LEFT OUTER JOIN (
    SELECT
      customer_store_month
      , SUM(__orders) AS customer_store_month__orders
    FROM sma_10003_cte
    GROUP BY
      customer_store_month
  ) subq_11
  ON
    subq_5.customer_store_month = subq_11.customer_store_month
  LEFT OUTER JOIN (
    SELECT
      store_id AS store
      , store_name
    FROM "jaffle_comparison"."main"."stores" stores_src_10000
  ) subq_13
  ON
    subq_5.store = subq_13.store
) subq_15
WHERE customer_store_month__orders > 10
GROUP BY
  metric_time__month
  , store__store_name
ORDER BY metric_time__month
